**Goal:** Decide whether the RQ0b change happened around 1970, around 1990, or gradually, using a rule fixed in advance. The decision covers total growth (annual, with oil-year effects) and net migration (by decade), and the growth result is split exactly into a moving part and a births-minus-deaths part.

**Architecture:**
- `modeling/annual_breaks.py` learns fixed-effect "cells". A cell is a year, or a year × group of states. Demeaning within cells removes them exactly, as Plan 1b did for years.
- On top of that sits a three-period test (before 1970, 1970–89, 1990 on). It produces the share of the way 1970–89 had moved toward "after", a unit-bootstrap range for that share, and a decision function.
- A new source, `sources/netmig.py`, reads county net migration by decade (UW–Madison APL / ICPSR). `features.py` sums it to states into a decade panel with total, migration and natural-increase rates.
- A new notebook 1.1 section runs everything and writes two tables and one figure.

**Tech Stack:** Python 3.12, pandas, numpy, scipy (chi-square p-values), statsmodels (the reference in tests only), matplotlib, pytest, jupytext and nbconvert.

**Spec:** `docs/superpowers/specs/2026-10-15-rq0b-1970-vs-1990-design.md` (approved 2026-10-15). It builds on the Plan 1b spec and code (`modeling/annual_breaks.py`).

- **The decision rule is fixed (spec §4.1); never tune it to results.**
  - `share = (β_pre − β_mid) / (β_pre − β_post)`, with a 90% range from resampling states (999 draws).
  - **"1970"** if low > 0 and high ≥ 1; **"1990"** if low ≤ 0 and high < 1; **"gradual"** if low > 0 and high < 1; **"can't tell"** if low ≤ 0 and high ≥ 1.
  - **Guard:** if β_pre = β_post isn't rejected at 5% (clustered Wald), the answer is "can't tell".
  - The main version sets the answer. If any robustness version gives a different answer, the finding is reported as "fragile".
- **Periods:** cuts at 1970 and 1990.
  - Annual data: before = growth years up to 1969, middle = 1970–1989, after = 1990 on.
  - Decade data: before = 1950s and 1960s, middle = 1970s and 1980s, after = 1990s on.

- **Wording (D-29):** findings say "growth followed income" or "net migration followed income". They say "moving" only for the net-migration results.

import statsmodels.api as sm


def state_panel(beta_for, n_units=48, years=range(1929, 2024), noise=0.02, seed=0, oil=False):
    """Like simulate(), with state-style IDs (state:01 ...) so oil and division cells apply. With
    oil=True, units 01-08 get an income boom 1973-81 and bust 1982-90, and growth follows oil jobs
    (+0.03 boom, -0.03 bust) rather than income."""
    rng = np.random.default_rng(seed)
    base = rng.normal(0, 0.3, n_units)
    energy = np.arange(n_units) < 8
    rows = []
    for y in years:
        cycle = (0.3 if 1973 <= y <= 1981 else -0.3 if 1982 <= y <= 1990 else 0.0) if oil else 0.0
        x = base + rng.normal(0, 0.05, n_units) + energy * cycle
        g = 0.01 * np.sin(y) + beta_for(y) * x + energy * np.sign(cycle) * 0.03 + rng.normal(0, noise, n_units)
        rows.append(pd.DataFrame({"geo_id": [f"state:{i + 1:02d}" for i in range(n_units)], "year": y,
                                  "log_income": x, "growth": g}))
    return pd.concat(rows, ignore_index=True)

def _cell_codes(dp: pd.DataFrame, period: str, cell: str | None) -> tuple[np.ndarray, int]:
    """Each row's fixed-effect cell (the period itself when cell is None) and the number of cells.
    A cell must lie inside one period, so demeaning within cells keeps the period slopes exact."""
    col = dp[period] if cell is None else dp[cell]
    codes, uniques = pd.factorize(col, sort=True)
    if cell is not None and dp.groupby(cell)[period].nunique().max() > 1:
        raise ValueError(f"cells in {cell!r} span more than one {period}")
    return codes, len(uniques)


def oil_cells(dp: pd.DataFrame, energy_fips, years: tuple[int, int] = OIL_YEARS, period: str = "year") -> pd.DataFrame:
    """cell = the period, except energy states during the oil years, which get their own cell (e.g.
    "1980E"), so the oil boom and bust can't show up as a change in beta."""
    in_oil = dp.geo_id.str[6:8].isin(set(energy_fips)) & dp[period].between(*years)
    return dp.assign(cell=dp[period].astype(str) + np.where(in_oil, "E", ""))


def division_cells(dp: pd.DataFrame, divisions: pd.Series, period: str = "year") -> pd.DataFrame:
    """cell = period x census division: states are compared only within their own region."""
    division = dp.geo_id.str[6:8].map(divisions)
    if division.isna().any():
        raise ValueError(f"no division for {sorted(dp.geo_id[division.isna()].unique())}")
    return dp.assign(cell=dp[period].astype(str) + "-" + division.astype(int).astype(str))
```

def sup_wald_break(dp: pd.DataFrame, trim: float = 0.15, cell: str | None = None) -> SingleBreak:
    """Break year = lowest SSR over candidates (Bai-Perron); significance = the largest clustered
    Wald statistic over the same candidates, against Andrews' critical value. cell: an optional
    fixed-effect column finer than the year (oil_cells, division_cells)."""
    periods, t, g, x, y = _arrays(dp)
    cc, ncell = _cell_codes(dp, "year", cell)
    nt, ones = len(periods), np.ones(len(t))
    xd, yd = _demean(cc, x, y, ones, ncell)
    cand = candidate_positions(nt, trim)
    ssr, b1, b2 = _ssr_candidates(*_sums(t, xd, yd, ones, nt), cand)
    wald = [_cluster_wald(t, g, xd, yd, c, s1, s2, ncell) for c, s1, s2 in zip(cand, b1, b2)]
    j = int(np.nanargmin(ssr))
    return SingleBreak(
        break_year=int(periods[cand[j]]), at_edge=j == 0, sup_wald=float(max(wald)),
        critical_value=andrews_critical_value(trim), beta_before=float(b1[j]), beta_after=float(b2[j]),
        trim=trim, first_candidate=int(periods[cand[0]]), last_candidate=int(periods[cand[-1]]),
        n_units=int(g.max() + 1), first_year=int(periods[0]), last_year=int(periods[-1]),
    )
```

```python
def _estimate(cc, ncell, t, x, y, w, nt, cand) -> int:
    """Position of the best-fitting break with rows weighted by w (demeaned within cells)."""
    xd, yd = _demean(cc, x, y, w, ncell)
    ssr, _, _ = _ssr_candidates(*_sums(t, xd, yd, w, nt), cand)
    return int(cand[int(np.nanargmin(ssr))])


def bootstrap_break_year(dp: pd.DataFrame, trim: float = 0.15, draws: int = 999,
                         level: float = 0.90, seed: int = 0, cell: str | None = None) -> BreakRange:
    """Resample whole units with replacement (keeping each unit's years together), re-estimate
    the break year each time, and return the middle `level` share of the estimates."""
    periods, t, g, x, y = _arrays(dp)
    cc, ncell = _cell_codes(dp, "year", cell)
    nt, n_units = len(periods), g.max() + 1
    cand = candidate_positions(nt, trim)
    rng = np.random.default_rng(seed)
    est = np.empty(draws, dtype=int)
    for d in range(draws):
        w = np.bincount(rng.integers(0, n_units, n_units), minlength=n_units)[g].astype(float)
        est[d] = periods[_estimate(cc, ncell, t, x, y, w, nt, cand)]
    a = (1 - level) / 2
    return BreakRange(int(np.floor(np.quantile(est, a))), int(np.ceil(np.quantile(est, 1 - a))), est)
```

```python
@dataclass(frozen=True)
class ThreePeriod:
    beta_pre: float  # before cuts[0]
    beta_mid: float  # cuts[0] .. cuts[1] - 1
    beta_post: float  # from cuts[1]
    share: float  # (pre - mid) / (pre - post): 0 = middle looks like before, 1 = like after
    p_pre_post: float  # the guard: did anything change at all?
    p_mid_pre: float
    p_mid_post: float
    wald_pre_post: float
    wald_mid_pre: float
    wald_mid_post: float
    n_units: int
    n_obs: int


def _segments(dp: pd.DataFrame, period: str, cuts) -> np.ndarray:
    return np.searchsorted(np.asarray(cuts), dp[period].to_numpy(), side="right")


def _three_betas(seg, xd, yd, w):
    sxx, sxy = np.bincount(seg, w * xd * xd, 3), np.bincount(seg, w * xd * yd, 3)
    with np.errstate(divide="ignore", invalid="ignore"):
        return sxy / sxx, sxx


def _share(b) -> float:
    with np.errstate(divide="ignore", invalid="ignore"):
        return float((b[0] - b[1]) / (b[0] - b[2]))


def three_period(dp: pd.DataFrame, cuts=RQ0B_CUTS, period: str = "year", cell: str | None = None) -> ThreePeriod:
    """Income slope before cuts[0], between the cuts, and from cuts[1], with fixed effects for
    each cell (the period by default), standard errors clustered by unit, statsmodels'
    small-sample factor, and chi-square(1) p-values (as wald_test(..., use_f=False))."""
    cc, ncell = _cell_codes(dp, period, cell)
    g = pd.factorize(dp.geo_id)[0]
    x, y = dp.log_income.to_numpy(float), dp.growth.to_numpy(float)
    ones = np.ones(len(x))
    xd, yd = _demean(cc, x, y, ones, ncell)
    seg = _segments(dp, period, cuts)
    b, sxx = _three_betas(seg, xd, yd, ones)
    e = yd - b[seg] * xd
    n_units, n, k = g.max() + 1, len(x), ncell + 3
    scores = np.column_stack([np.bincount(g, np.where(seg == s, xd * e, 0.0), n_units) for s in range(3)])
    cov = n_units / (n_units - 1) * (n - 1) / (n - k) * (scores.T @ scores) / np.outer(sxx, sxx)

    def wald(a):
        a = np.asarray(a, float)
        stat = float((a @ b) ** 2 / (a @ cov @ a))
        return stat, float(chi2.sf(stat, 1))

    (w_pp, p_pp), (w_mpre, p_mpre), (w_mpost, p_mpost) = wald([1, 0, -1]), wald([-1, 1, 0]), wald([0, 1, -1])
    return ThreePeriod(float(b[0]), float(b[1]), float(b[2]), _share(b), p_pp, p_mpre, p_mpost,
                       w_pp, w_mpre, w_mpost, int(n_units), int(n))


def share_range(dp: pd.DataFrame, cuts=RQ0B_CUTS, period: str = "year", cell: str | None = None,
                draws: int = 999, level: float = 0.90, seed: int = 0) -> tuple[float, float, np.ndarray]:
    """Middle `level` range of the share over unit-bootstrap draws (whole units resampled, years
    kept together). Draws where a period has no x-variation give NaN and are skipped."""
    cc, ncell = _cell_codes(dp, period, cell)
    g = pd.factorize(dp.geo_id)[0]
    n_units = g.max() + 1
    x, y = dp.log_income.to_numpy(float), dp.growth.to_numpy(float)
    seg = _segments(dp, period, cuts)
    rng = np.random.default_rng(seed)
    shares = np.empty(draws)
    for d in range(draws):
        w = np.bincount(rng.integers(0, n_units, n_units), minlength=n_units)[g].astype(float)
        xd, yd = _demean(cc, x, y, w, ncell)
        shares[d] = _share(_three_betas(seg, xd, yd, w)[0])
    a = (1 - level) / 2
    return float(np.nanquantile(shares, a)), float(np.nanquantile(shares, 1 - a)), shares


def decide(low: float, high: float, p_pre_post: float, cuts=RQ0B_CUTS, alpha: float = 0.05) -> str:
    """The Plan 1c rule, fixed before running (spec §4.1): which period 1970-89 resembles."""
    if not p_pre_post < alpha:
        return "can't tell"
    if low > 0 and high >= 1:
        return str(cuts[0])
    if low <= 0 and high < 1:
        return str(cuts[1])
    if low > 0 and high < 1:
        return "gradual"
    return "can't tell"
```

"""County net migration by decade: the UW-Madison Applied Population Laboratory series (Winkler,
Johnson, Curtis and others), archived at ICPSR as the "United States County-Specific Net
Migration Rates Series" (collection 2760). The 2010s file downloads from netmigration.wisc.edu;
the 1950s-2000s files need a free ICPSR login and no redistribution, so they are downloaded by
hand into data/raw/netmig/ (like ICPSR 2896) and read straight from the zips. Net migration is
the residual method's estimate: population change minus births plus deaths."""

import io
from pathlib import Path
import zipfile

import pandas as pd

from us_housing_crisis_analysis.config import NETMIG_FILES, RAW_DATA_DIR
from us_housing_crisis_analysis.schema import validate_long
from us_housing_crisis_analysis.sources._http import download_file
from us_housing_crisis_analysis.sources.icpsr import MissingManualDownloadError

RAW_DIR = RAW_DATA_DIR / "netmig"
APL_URLS = {2010: "https://netmigration.wisc.edu/documents/NME_1020_data.zip"}
SERIES_URL = "https://www.icpsr.umich.edu/sites/icpsr/view/collections/2760"

__all__ = ["MissingManualDownloadError", "download", "load", "parse_net_migration"]


def download(force: bool = False) -> list[Path]:
    paths = []
    for decade, spec in NETMIG_FILES.items():
        dest = RAW_DIR / spec["file"]
        if decade in APL_URLS:
            download_file(APL_URLS[decade], dest, force=force)
        paths.append(dest)
    missing = [p.name for p in paths if not p.exists()]
    if missing:
        raise MissingManualDownloadError(
            f"missing {missing} in {RAW_DIR}. The ICPSR decades need a free login:\n"
            f"  1. Open {SERIES_URL} and log in.\n"
            "  2. For each decade, download the county file (delimited or Stata) with its codebook.\n"
            f"  3. Save the zips, unopened, in {RAW_DIR} under the names in config.NETMIG_FILES."
        )
    return paths


def _read(path: Path, member: str) -> pd.DataFrame:
    with zipfile.ZipFile(path) as zf:
        raw = zf.read(member)
    suffix = Path(member).suffix.lower()
    if suffix == ".dta":
        return pd.read_stata(io.BytesIO(raw), convert_categoricals=False)
    return pd.read_csv(io.BytesIO(raw), sep="\t" if suffix in (".tsv", ".tab", ".txt") else ",", dtype=str)


def parse_net_migration(path: Path, decade: int, spec: dict[str, str]) -> pd.DataFrame:
    raw = _read(path, spec["member"])
    fips = raw[spec["fips_col"]].astype(str).str.strip().str.replace(r"\.0$", "", regex=True).str.zfill(5)
    cols = spec["total_col"].split("+")
    value = raw[cols].apply(pd.to_numeric, errors="coerce").sum(axis=1, min_count=len(cols))
    # County rows only: xx000 rows are state totals and 00000 the US; never add them to counties
    keep = fips.str.fullmatch(r"\d{5}") & ~fips.str.endswith("000") & value.notna()
    return pd.DataFrame(
        {
            "geo_id": "county:" + fips[keep], "geo_name": raw.loc[keep, spec["name_col"]].astype(str),
            "geo_level": "county", "country_iso3": "USA", "year": decade, "variable": "net_migration",
            "value": value[keep].astype("float64"), "unit": "persons", "source": "netmig",
        }
    ).astype({"year": "int64"}).reset_index(drop=True)


def load() -> pd.DataFrame:
    paths = download()
    frames = [parse_net_migration(p, d, spec) for p, (d, spec) in zip(paths, NETMIG_FILES.items())]
    return validate_long(pd.concat(frames, ignore_index=True))

def build_us_state_net_migration(longrun: pd.DataFrame, netmig: pd.DataFrame) -> pd.DataFrame:
    """State x decade (Plan 1c): census population at both ends, income at the start, county net
    migration summed to the state. Rates are simple percent changes of the starting population,
    so total = migration + natural increase exactly. A state missing net migration for a decade
    drops out of that decade only."""
    nm = netmig[(netmig.variable == "net_migration") & (netmig.geo_level == "county")]
    nm = (nm.assign(geo_id="state:" + nm.geo_id.str[7:9]).groupby(["geo_id", "year"], as_index=False)["value"].sum()
          .rename(columns={"value": "net_migration"}))
    p = longrun[["geo_id", "geo_name", "year", "population", "income_pc"]]
    nxt = p[["geo_id", "year", "population"]].assign(year=p.year - 10).rename(columns={"population": "population_next"})
    m = p.merge(nxt, on=["geo_id", "year"]).merge(nm, on=["geo_id", "year"])
    m = m[m.year.isin(list(NETMIG_DECADES)) & (m.population > 0)]
    total = (m.population_next - m.population) / m.population
    migration = m.net_migration / m.population
    out = m.rename(columns={"year": "decade"}).assign(
        total_rate=total, migration_rate=migration, natural_rate=total - migration)
    cols = ["geo_id", "geo_name", "decade", "population", "population_next", "income_pc", "net_migration",
            "total_rate", "migration_rate", "natural_rate"]
    return out[cols].sort_values(["geo_id", "decade"]).reset_index(drop=True)
